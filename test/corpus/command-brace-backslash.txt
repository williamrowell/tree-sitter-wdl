==========================================
Backslashes in brace-style commands
==========================================

version 1.0

task Brace {
    input {
        String x
    }
    command {
        echo a\nb ~{x}\t
        cut -d$'\t' -f1 file | sed 's/a\/b//'
        echo a \
            b \\ c \} d
    }
}
