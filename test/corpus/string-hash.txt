==========================================
A hash at the start of string content is not a comment
==========================================

version 1.0

task Hash {
    input {
        String x
        Boolean b
    }
    String a = "#"
    String c = '#x'
    String d = "v~{x}#tag"
    command <<<
        echo ~{true="" false="#" b} # trailing comment
    >>>
}
